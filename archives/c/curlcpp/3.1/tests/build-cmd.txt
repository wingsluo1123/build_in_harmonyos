# 功能测试编译命令 (静态链接 curlcpp + libcurl + openssl + zlib)
PKG_DIR=$(find $HOME/.conan2/p -path "*curlc*" -type d -name "p" 2>/dev/null | head -1)

clang++ -stdlib=libc++ -std=c++11 \
    -I$PKG_DIR/include \
    -I$HOME/usr/local/include \
    curlcpp_func_test.cpp \
    $PKG_DIR/lib/libcurlcpp.a \
    $HOME/usr/local/lib/libcurl.a \
    $HOME/usr/local/lib/libssl.a \
    $HOME/usr/local/lib/libcrypto.a \
    $HOME/usr/local/lib/libz.a \
    -lpthread -ldl \
    -o curlcpp_func_test
