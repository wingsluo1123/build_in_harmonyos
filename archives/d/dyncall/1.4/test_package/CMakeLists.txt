cmake_minimum_required(VERSION 3.15)
project(dyncall_consumer LANGUAGES C)
find_package(dyncall CONFIG REQUIRED)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
add_executable(consumer consumer.c)
target_link_libraries(consumer PRIVATE dyncall::dyncallback dyncall::dyncall dyncall::dynload)
