cmake_minimum_required(VERSION 3.15)
project(test_gst_plugins_ugly C)

# Find packages via Conan-generated CMakeDeps
find_package(gst-plugins-ugly REQUIRED)
find_package(gst-plugins-base REQUIRED)
find_package(gstreamer REQUIRED)
find_package(glib2 REQUIRED)

add_executable(test_gst_plugins_ugly test_gst_plugins_ugly.c)
target_link_libraries(test_gst_plugins_ugly
    gst-plugins-ugly::gst-plugins-ugly
    gstreamer::gstreamer
    glib2::glib2
)

# Workaround: Conan 2.29.1 CMakeDeps does not propagate includedirs set in
# package_info() when using basic_layout. Add include paths manually using
# the PACKAGE_FOLDER variables generated by CMakeDeps.
if(DEFINED gstreamer_PACKAGE_FOLDER_RELEASE)
    target_include_directories(test_gst_plugins_ugly PRIVATE
        ${gstreamer_PACKAGE_FOLDER_RELEASE}/include/gstreamer-1.0)
elseif(DEFINED gstreamer_PACKAGE_FOLDER_DEBUG)
    target_include_directories(test_gst_plugins_ugly PRIVATE
        ${gstreamer_PACKAGE_FOLDER_DEBUG}/include/gstreamer-1.0)
endif()

if(DEFINED glib2_PACKAGE_FOLDER_RELEASE)
    target_include_directories(test_gst_plugins_ugly PRIVATE
        ${glib2_PACKAGE_FOLDER_RELEASE}/include/glib-2.0
        ${glib2_PACKAGE_FOLDER_RELEASE}/lib/glib-2.0/include)
elseif(DEFINED glib2_PACKAGE_FOLDER_DEBUG)
    target_include_directories(test_gst_plugins_ugly PRIVATE
        ${glib2_PACKAGE_FOLDER_DEBUG}/include/glib-2.0
        ${glib2_PACKAGE_FOLDER_DEBUG}/lib/glib-2.0/include)
endif()

enable_testing()
add_test(NAME test_gst_plugins_ugly COMMAND test_gst_plugins_ugly)
