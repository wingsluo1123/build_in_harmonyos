cmake_minimum_required(VERSION 3.21)
project(real_parallel_hdf5_consumer C)
find_package(MPICH CONFIG REQUIRED)
find_package(HDF5 CONFIG REQUIRED)
if(NOT HDF5_IS_PARALLEL)
  message(FATAL_ERROR "Actual parallel HDF5 producer capability missing")
endif()
add_executable(real_parallel main.c)
target_link_libraries(real_parallel PRIVATE hdf5-static MPI::MPI_C)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  add_subdirectory(harmony_tests)
endif()
