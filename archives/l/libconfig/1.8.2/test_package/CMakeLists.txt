cmake_minimum_required(VERSION 3.15)
project(libconfig_consumer LANGUAGES C CXX)
find_package(libconfig CONFIG REQUIRED)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
add_executable(consumer_c consumer.c)
target_link_libraries(consumer_c PRIVATE libconfig::config)
add_executable(consumer_cpp consumer.cpp)
target_link_libraries(consumer_cpp PRIVATE libconfig::config++)
target_compile_features(consumer_cpp PRIVATE cxx_std_17)
