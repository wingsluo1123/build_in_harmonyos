cmake_minimum_required(VERSION 3.22)
project(test_package LANGUAGES CXX)

find_package(KPim6Libkleo QUIET CONFIG)
if(NOT KPim6Libkleo_FOUND)
    find_package(KPimLibkleo REQUIRED CONFIG)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core)

add_executable(test_libkleo test_libkleo.cpp)
target_compile_features(test_libkleo PRIVATE cxx_std_20)
if(TARGET KPim6::Libkleo)
    set(_libkleo_target KPim6::Libkleo)
elseif(TARGET KPimLibkleo::Libkleo)
    set(_libkleo_target KPimLibkleo::Libkleo)
elseif(TARGET KPim6Libkleo::KPim6Libkleo)
    set(_libkleo_target KPim6Libkleo::KPim6Libkleo)
else()
    message(FATAL_ERROR "libkleo target not found")
endif()
target_link_libraries(test_libkleo PRIVATE ${_libkleo_target} Qt6::Core)

enable_testing()
add_test(NAME test_libkleo COMMAND $<TARGET_FILE:test_libkleo>)
