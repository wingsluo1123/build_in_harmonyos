cmake_minimum_required(VERSION 3.16)
project(quorum VERSION 1.1.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(jellyfish CONFIG REQUIRED)
find_package(ZLIB CONFIG REQUIRED)
find_package(GTest CONFIG REQUIRED)
find_package(Threads REQUIRED)
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("int main(){ __int128 x=1; return int(x); }" HAVE_INT128)
check_cxx_source_compiles("#include <limits>\nstatic_assert(std::numeric_limits<__int128>::is_specialized, \"no specialization\"); int main(){}" HAVE_NUMERIC_LIMITS128)
configure_file(config.h.cmake config.h)
add_library(quorum_options INTERFACE)
target_include_directories(quorum_options INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(quorum_options INTERFACE PACKAGE_VERSION="${PROJECT_VERSION}")
target_link_libraries(quorum_options INTERFACE jellyfish::jellyfish ZLIB::ZLIB Threads::Threads)
add_executable(quorum_error_correct_reads src/error_correct_reads.cc src/err_log.cc)
add_executable(quorum_create_database src/create_database.cc)
add_executable(merge_mate_pairs src/merge_mate_pairs.cc)
add_executable(split_mate_pairs src/split_mate_pairs.cc)
foreach(target quorum_error_correct_reads quorum_create_database merge_mate_pairs split_mate_pairs)
  target_link_libraries(${target} PRIVATE quorum_options)
endforeach()
add_executable(all_tests unit_tests/test_mer_database.cc)
target_link_libraries(all_tests PRIVATE quorum_options GTest::gtest_main)
add_executable(query_mer_database src/query_mer_database.cc)
add_executable(histo_mer_database src/histo_mer_database.cc)
target_link_libraries(query_mer_database PRIVATE quorum_options)
target_link_libraries(histo_mer_database PRIVATE quorum_options)
enable_testing()
add_test(NAME all_upstream_mer_database_instances COMMAND all_tests)
set(JF_LIB_PATH "")
set(bindir "$dir")
set(JELLYFISH "jellyfish")
set(PACKAGE_VERSION "${PROJECT_VERSION}")
set(JELLYFISH_VERSION "2.3.1")
configure_file(src/quorum.in quorum @ONLY)
install(TARGETS quorum_error_correct_reads quorum_create_database merge_mate_pairs split_mate_pairs RUNTIME DESTINATION bin)
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/quorum" DESTINATION bin)
install(FILES data/adapter.jf DESTINATION share/quorum)
