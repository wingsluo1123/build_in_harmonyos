# ---------------------------------------------------------------------------
# signond — Single Sign On daemon (accounts-sso), HarmonyOS PC adaptation.
#
# Upstream ships a qmake-only build (signon.pro + *.pri); the ohpcd qtbase
# package has no OHOS mkspec, so qmake cannot cross-compile.  This hand-written
# CMakeLists reproduces the qmake targets of the *daemon* half of the source
# tree (lib/plugins, lib/signond, src/) for the Qt6 flavour:
#
#   lib/plugins/signon-plugins.pro  -> signon-plugins        (SDK carrier lib)
#   lib/plugins/signon-plugins-common/signon-plugins-common.pro
#                                   -> signon-plugins-common (blob IPC helpers)
#   lib/signond/SignOn/SignOnExtension.pro
#                                   -> signon-extension      (extension SDK)
#   src/signond/signond.pro         -> signond               (the daemon)
#   src/remotepluginprocess/remotepluginprocess.pro
#                                   -> signonpluginprocess   (plugin host)
#   src/plugins/password/password.pro
#                                   -> passwordplugin        (auth plugin)
#   src/plugins/example/example.pro -> exampleplugin         (sample plugin)
#
# The client library half (lib/SignOn) is published separately as the
# `libsignon-qt` package and is intentionally NOT rebuilt here.
# ---------------------------------------------------------------------------

# common-project-config.pri: SIGNOND_PLUGINS_DIR / SIGNOND_EXTENSIONS_DIR are
# baked into the binaries as the *default* extension search path; the daemon
# honours SSO_PLUGINS_DIR / SSO_EXTENSIONS_DIR at runtime (signondaemon.cpp).
# Upstream defaults to $INSTALL_LIBDIR/signon — a stable system path, so no
# conan cache path leaks into the ELF files.
set(SIGNOND_PLUGINS_DIR "/usr/${CMAKE_INSTALL_LIBDIR}/signon"
    CACHE PATH "Compile-time default signond plugin directory")
set(SIGNOND_EXTENSIONS_DIR "/usr/${CMAKE_INSTALL_LIBDIR}/signon/extensions"
    CACHE PATH "Compile-time default signond extension directory")

# plugins link these SDK libraries, so shared is the default (upstream builds

    set(SIGNOND_LIB_TYPE STATIC)
endif()

# DT_SONAME for the SDK shared libraries.
#
# Upstream's qmake projects never set a VERSION, so upstream .so files carry no
# DT_SONAME; the repository's L5a CI gate requires one (a downstream linker
# cannot address a library by soname without it).  CMake cannot supply it
# either on this platform: CMAKE_SYSTEM_NAME=HarmonyOS is unknown to CMake, so
# CMAKE_SHARED_LIBRARY_SONAME_CXX_FLAG is empty and CMake's VERSION/SOVERSION
# properties emit no -soname at all.  Pass the soname explicitly, using the
# unversioned library name (which is also what upstream installs, and what the
# daemon/plugins resolve at runtime).
function(signond_set_soname target soname)
    if(BUILD_SHARED_LIBS AND NOT MSVC)
        target_link_options(${target} PRIVATE "LINKER:-soname,${soname}")
    endif()
endfunction()

# ---------------------------------------------------------------------------
# signon-plugins — lib/plugins/signon-plugins.pro
#
# Upstream declares TEMPLATE=lib with HEADERS only (SignOn/authpluginif.h,
# SignOn/uisessiondata.h, SignOn/uisessiondata_priv.h).  qmake runs moc over
# authpluginif.h and compiles the generated code into libsignon-plugins.so, so
# the library is what supplies AuthPluginInterface's metacast/metaobject/vtable
# to every plugin (src/plugins/plugins.pri: LIBS += -lsignon-plugins).  Include
# the SDK headers from a real translation unit so CMAKE_AUTOMOC reproduces that
# moc output instead of shipping an empty carrier library.
# ---------------------------------------------------------------------------
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/signon-plugins-sdk.cpp"
"// Generated by the HarmonyOS CMake build: upstream signon-plugins.pro builds
// SignOn/authpluginif.h through moc into libsignon-plugins.  Including the SDK
// headers here makes CMAKE_AUTOMOC generate and compile moc_authpluginif.cpp
// for this target, which is where every plugin resolves
// AuthPluginInterface::qt_metacast/qt_metacall/staticMetaObject from.
#include <SignOn/authpluginif.h>
#include <SignOn/uisessiondata.h>
#include <SignOn/uisessiondata_priv.h>
")
add_library(signon-plugins ${SIGNOND_LIB_TYPE}
    "${CMAKE_CURRENT_BINARY_DIR}/signon-plugins-sdk.cpp"
    # Listing the SDK headers as sources is what makes CMAKE_AUTOMOC run moc over
    # them (AUTOMOC scans target sources, it does not follow #include chains);
    # the generated moc_authpluginif.cpp goes into libsignon-plugins.
    "${SIGNOND_TOP}/lib/plugins/SignOn/authpluginif.h"
    "${SIGNOND_TOP}/lib/plugins/SignOn/uisessiondata.h"
    "${SIGNOND_TOP}/lib/plugins/SignOn/uisessiondata_priv.h"
)
target_include_directories(signon-plugins
    PUBLIC
        $<BUILD_INTERFACE:${SIGNOND_TOP}/lib/plugins>
        $<BUILD_INTERFACE:${SIGNOND_TOP}/lib>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/signon-plugins>
)
target_link_libraries(signon-plugins PUBLIC Qt6::Core Qt6::DBus)
