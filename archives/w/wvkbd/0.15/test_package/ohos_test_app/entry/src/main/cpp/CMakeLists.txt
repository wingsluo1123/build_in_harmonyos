cmake_minimum_required(VERSION 3.16)
project(wvkbd_device_consumer LANGUAGES C CXX)
if(NOT DEFINED wvkbd_DIR AND DEFINED ENV{WVKBD_CONAN_GENERATORS} AND
   NOT "$ENV{WVKBD_CONAN_GENERATORS}" STREQUAL "")
    set(wvkbd_DIR "$ENV{WVKBD_CONAN_GENERATORS}" CACHE PATH "Conan CMakeDeps directory")
endif()
find_package(wvkbd CONFIG REQUIRED NO_CMAKE_FIND_ROOT_PATH)
if(NOT EXISTS "$ENV{WVKBD_TEST_CONSUMER_LIBRARY}")
    message(FATAL_ERROR "WVKBD_TEST_CONSUMER_LIBRARY must name the Conan-built consumer archive")
endif()
add_library(wvkbd_test_consumer STATIC IMPORTED)
set_target_properties(wvkbd_test_consumer PROPERTIES IMPORTED_LOCATION "$ENV{WVKBD_TEST_CONSUMER_LIBRARY}")
add_library(wvkbd_test SHARED test_napi.cpp)
set_target_properties(wvkbd_test PROPERTIES SKIP_BUILD_RPATH ON)
target_link_libraries(wvkbd_test PRIVATE wvkbd_test_consumer wvkbd::core libace_napi.z.so)
target_compile_features(wvkbd_test PRIVATE cxx_std_17)
target_compile_options(wvkbd_test PRIVATE -Wall -Wextra -Werror -Wno-unused-command-line-argument)
