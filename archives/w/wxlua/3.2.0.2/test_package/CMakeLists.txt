cmake_minimum_required(VERSION 3.15)
project(wxlua_consumer LANGUAGES C CXX)
find_package(wxlua CONFIG REQUIRED)
find_package(lua CONFIG REQUIRED)
add_executable(consumer consumer.cpp)
target_link_libraries(consumer PRIVATE wxlua::wxlua)
add_executable(module_host module_host.c)
target_link_libraries(module_host PRIVATE "-Wl,--whole-archive" lua::lua "-Wl,--no-whole-archive" dl m)
target_link_options(module_host PRIVATE "-Wl,--export-dynamic")
